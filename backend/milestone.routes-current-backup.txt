
router.post("/:id/submissions", async (req, res) => {
  let submissionId: mongoose.Types.ObjectId | undefined;

  try {
    const { id } = req.params;

    // 1. Validate milestone ID
    if (!mongoose.isValidObjectId(id)) {
      return sendError(res, 400, "INVALID_ID", "Invalid milestone ID");
    }

    // 2. Validate request body
    const parsed = createSubmissionSchema.safeParse(req.body);

    if (!parsed.success) {
      return sendError(
        res,
        400,
        "VALIDATION_ERROR",
        "Invalid submission data",
        parsed.error.issues
      );
    }

    // 3. Load milestone
    const milestone = await Milestone.findById(id);

    if (!milestone) {
      return sendError(res, 404, "NOT_FOUND", "Milestone not found");
    }

    // 4. Load associated project
    const project = await Project.findById(milestone.projectId);

    if (!project) {
      return sendError(res, 404, "NOT_FOUND", "Milestone not found");
    }

    // 5. Check project participation
    const userId = req.user!._id.toString();

    const isClient = project.clientId.toString() === userId;

    const isFreelancer =
      project.freelancerId?.toString() === userId;

    if (!isClient && !isFreelancer) {
      return sendError(res, 404, "NOT_FOUND", "Milestone not found");
    }

    // 6. Check freelancer role
    if (req.user!.role !== "freelancer") {
      return sendError(
        res,
        403,
        "FORBIDDEN",
        "Only freelancers can submit work"
      );
    }

    // 7. Confirm assigned freelancer
    if (!isFreelancer) {
      return sendError(
        res,
        403,
        "FORBIDDEN",
        "Only the assigned freelancer can submit this milestone"
      );
    }

    // 8. Check milestone status
    const allowedStatuses = [
      "criteria_approved",
      "in_progress",
      "rejected",
    ] as const;

    if (!allowedStatuses.includes(milestone.status as typeof allowedStatuses[number])) {
      return sendError(
        res,
        409,
        "INVALID_STATUS",
        "Milestone cannot be submitted in its current status"
      );
    }

    // 9. Create submission
    const submission = await Submission.create({
      milestoneId: milestone._id,
      projectId: project._id,
      freelancerId: req.user!._id,
      description: parsed.data.description,
      previewUrl: parsed.data.previewUrl,
      source: "freelancer_claim",
    });

    submissionId = submission._id;

    // 10. Update milestone status atomically
    const updatedMilestone = await Milestone.findOneAndUpdate(
      {
        _id: id,
        status: {
          $in: ["criteria_approved", "in_progress", "rejected"],
        },
      },
      {
        $set: { status: "submitted" },
      },
      {
        new: true,
        runValidators: true,
      }
    );

    if (!updatedMilestone) {
      await Submission.findByIdAndDelete(submission._id);

      return sendError(
        res,
        409,
        "INVALID_STATUS",
        "Milestone status changed. Please try again."
      );
    }

    // 11. Return successful response
    return res.status(201).json({
      message: "Submission created successfully",
      submission,
      milestone: updatedMilestone,
    });

  } catch (error) {
    if (submissionId) {
      await Submission.findByIdAndDelete(submissionId).catch(() => null);
    }

    console.error("Create submission error:", error);

    return sendError(
      res,
      500,
      "SERVER_ERROR",
      "Unable to create submission"
    );
  }
});
