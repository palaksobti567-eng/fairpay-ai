import mongoose from "mongoose";

export async function connectDB(): Promise<void> {
  const mongoUri = process.env.MONGODB_URI;

  if (!mongoUri) {
    console.error("MONGODB_URI is missing.");
    process.exit(1);
  }

  try {
    
    await mongoose.connect(mongoUri, {
      serverSelectionTimeoutMS: 10000,
    });

    console.log("MongoDB connected successfully");
  } catch (error) {
    console.error("MongoDB connection failed:");

    if (error instanceof Error) {
      console.error("Name:", error.name);
      console.error("Message:", error.message);

      const mongoError = error as any;

      if (mongoError.reason?.servers) {
        for (const [host, server] of mongoError.reason.servers) {
          console.error("Server:", host);
          console.error(
            "Server error:",
            server.error?.message || "No specific error"
          );
        }
      }
    } else {
      console.error(error);
    }

    process.exit(1);
  }
}

import { Request, Response, NextFunction } from "express";
import User, { UserRole, IUser } from "../models/User";
import { verifyToken } from "../utils/jwt";
import { sendError } from "../utils/errors";

declare global {
  namespace Express {
    interface Request {
      user?: IUser;
    }
  }
}

export async function requireAuth(
  req: Request,
  res: Response,
  next: NextFunction
) {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader || !authHeader.startsWith("Bearer ")) {
      return sendError(res, 401, "AUTH_REQUIRED", "Authentication required");
    }

    const token = authHeader.slice(7);
    const { userId } = verifyToken(token);

    const user = await User.findById(userId);

    if (!user) {
      return sendError(res, 401, "INVALID_TOKEN", "Invalid or expired token");
    }

    req.user = user;
    next();
  } catch {
    return sendError(res, 401, "INVALID_TOKEN", "Invalid or expired token");
  }
}

export function requireRole(role: UserRole) {
  return (req: Request, res: Response, next: NextFunction) => {
    if (!req.user) {
      return sendError(res, 401, "AUTH_REQUIRED", "Authentication required");
    }

    if (req.user.role !== role) {
      return sendError(res, 403, "FORBIDDEN", "Access forbidden");
    }

    next();
  };
}

import { Router, Request, Response } from "express";
import mongoose from "mongoose";
import { z } from "zod";
import Project from "../models/Project";
import { requireAuth, requireRole } from "../middleware/auth.middleware";
import { sendError } from "../utils/errors";
import {
  generateMilestones,
  AiGenerationError,
} from "../services/milestoneGenerator";

const router = Router();

const generateSchema = z.object({
  projectId: z
    .string()
    .refine((v) => mongoose.isValidObjectId(v), "Invalid project id"),

  description: z.string().trim().min(10).max(2000),
}).strict();

// POST /api/ai/generate-milestones
// Generates suggestions only. Nothing is saved.
router.post(
  "/generate-milestones",
  requireAuth,
  requireRole("client"),
  async (req: Request, res: Response) => {
    try {
      const parsed = generateSchema.safeParse(req.body);

      if (!parsed.success) {
        return sendError(
          res,
          400,
          "VALIDATION_ERROR",
          "Invalid request data",
          parsed.error.issues
        );
      }

      const { projectId, description } = parsed.data;

      // Only the project owner can generate milestones.
      const project = await Project.findOne({
        _id: projectId,
        clientId: req.user!._id,
      });

      if (!project) {
        return sendError(
          res, 404, "NOT_FOUND", "Project not found"
        );
      }

      // Budget must come from the saved project.
      if (
        !Number.isFinite(project.totalBudget) ||
        project.totalBudget <= 0
      ) {
        return sendError(
          res,
          409,
          "INVALID_STATE",
          "Project budget is not configured"
        );
      }

      const milestones = await generateMilestones({
        description,
        totalBudget: project.totalBudget,
        currency: project.currency,
      });

      return res.status(200).json({ milestones });

    } catch (err) {
      if (err instanceof AiGenerationError) {
        if (err.code === "AI_UNAVAILABLE") {
          return sendError(
            res, 502, "AI_UNAVAILABLE",
            "AI service is temporarily unavailable"
          );
        }

        return sendError(
          res, 422, "AI_INVALID_OUTPUT",
          "AI service returned an invalid milestone response"
        );
      }

      return sendError(
        res,
        500,
        "INTERNAL_ERROR",
        "Could not generate milestones"
      );
    }
  }
);

export default router;

import { Router } from "express";
import User from "../models/User";
import { hashPassword, comparePassword } from "../utils/password";

const router = Router();

// REGISTER
router.post("/register", async (req, res) => {
  try {
    const parsed = registerSchema.safeParse(req.body);

    if (!parsed.success) {
      return sendError(
        res,
        400,
        "VALIDATION_ERROR",
        "Invalid registration data",
        parsed.error.issues
      );
    }

    const { name, email, password } = parsed.data;

    const existingUser = await User.findOne({ email });

    if (existingUser) {
      return sendError(
        res,
        409,
        "EMAIL_TAKEN",
        "Email already registered"
      );
    }

    const passwordHash = await hashPassword(password);

    const user = await User.create({
      name,
      email,
      passwordHash,
      role: "client",
    });

    const token = generateToken(user._id.toString());

    return res.status(201).json({
      message: "Registration successful",
      token,
      user: {
        id: user._id,
        name: user.name,
        email: user.email,
        role: user.role,
      },
    });
  } catch (error: any) {
    if (error?.code === 11000) {
      return sendError(
        res,
        409,
        "EMAIL_TAKEN",
        "Email already registered"
      );
    }

    return sendError(
      res,
      500,
      "INTERNAL_ERROR",
      "Internal server error"
    );
  }
});

const MAX_SCREENSHOT_BYTES = 3 * 1024 * 1024;

function decodeScreenshot(value: unknown): Buffer {
  if (typeof value !== "string") {
    throw new Error("Invalid screenshot");
  }

  const buffer = Buffer.from(value, "base64");

  if (
    buffer.length === 0 ||
    buffer.length > MAX_SCREENSHOT_BYTES ||
    buffer.subarray(0, 8).toString("hex") !== "89504e470d0a1a0a"
  ) {
    throw new Error("Invalid screenshot");
  }

  return buffer;
}

function hasExactCriterionIds(
  expectedIds: string[],
  actualIds: string[]
): boolean {
  if (expectedIds.length !== actualIds.length) {

export class UnsafeUrlError extends Error {

function ipv4ToNumber(ip: string): number {
  return ip.split(".").reduce(
    (value, part) => value * 256 + Number(part),
    0
  );
}

function ipv4InCidr(ip: string, network: string, bits: number): boolean {
  const mask = bits === 0 ? 0 : (0xffffffff << (32 - bits)) >>> 0;

  return (
    (ipv4ToNumber(ip) & mask) ===
    (ipv4ToNumber(network) & mask)
  );
}

const BLOCKED_IPV4 = [
  ["0.0.0.0", 8],
  ["10.0.0.0", 8],
  ["100.64.0.0", 10],
  ["127.0.0.0", 8],
  ["169.254.0.0", 16],
  ["172.16.0.0", 12],
  ["192.0.0.0", 24],
  ["192.0.2.0", 24],
  ["192.88.99.0", 24],
  ["192.168.0.0", 16],
  ["198.18.0.0", 15],
  ["198.51.100.0", 24],
  ["203.0.113.0", 24],
  ["224.0.0.0", 4],
  ["240.0.0.0", 4],
] as const;

function ipv4IsUnsafe(ip: string): boolean {
  if (isIP(ip) !== 4) return true;

  return BLOCKED_IPV4.some(([network, bits]) =>
    ipv4InCidr(ip, network, bits)
  );
}

function ipv6ToBigInt(ip: string): bigint {
  let address = ip.toLowerCase();

  // Convert an embedded IPv4 address into two IPv6 groups.
  if (address.includes(".")) {
    const lastColon = address.lastIndexOf(":");
    const ipv4 = address.slice(lastColon + 1);

    if (isIP(ipv4) !== 4) throw new UnsafeUrlError();

    const value = ipv4ToNumber(ipv4);
    const first = ((value >>> 16) & 0xffff).toString(16);
    const second = (value & 0xffff).toString(16);

    address = address.slice(0, lastColon) + ":" + first + ":" + second;
  }

  const halves = address.split("::");

  if (halves.length > 2) throw new UnsafeUrlError();

  const left = halves[0] ? halves[0].split(":") : [];
  const right = halves.length === 2 && halves[1]
    ? halves[1].split(":")
    : [];

  const missing = 8 - left.length - right.length;

  if (
    (halves.length === 1 && missing !== 0) ||
    (halves.length === 2 && missing < 1)
  ) {
    throw new UnsafeUrlError();
  }

  const groups = [
    ...left,
    ...Array(missing).fill("0"),
    ...right,
  ];

  if (groups.length !== 8) throw new UnsafeUrlError();

  return groups.reduce((result, group) => {
    if (!/^[0-9a-f]{1,4}$/.test(group)) {
      throw new UnsafeUrlError();
    }

    return (result << 16n) | BigInt(parseInt(group, 16));
  }, 0n);
}

function ipv6InCidr(ip: string, network: string, bits: number): boolean {
  const mask = ((1n << BigInt(bits)) - 1n) << BigInt(128 - bits);

  return (
    (ipv6ToBigInt(ip) & mask) ===
    (ipv6ToBigInt(network) & mask)
  );
}

const BLOCKED_IPV6 = [
  ["2001::", 23],
  ["2001:2::", 48],
  ["2001:db8::", 32],
  ["2001:10::", 28],
  ["2001:20::", 28],
  ["2002::", 16],
  ["3fff::", 20],
] as const;

function ipv6IsUnsafe(ip: string): boolean {
  if (isIP(ip) !== 6) return true;

  // Only allow globally routable unicast IPv6 (2000::/3).
  if (!ipv6InCidr(ip, "2000::", 3)) return true;

  return BLOCKED_IPV6.some(([network, bits]) =>
    ipv6InCidr(ip, network, bits)
  );
}

function addressIsUnsafe(address: string): boolean {
  const family = isIP(address);

  if (family === 4) return ipv4IsUnsafe(address);
  if (family === 6) return ipv6IsUnsafe(address);

  return true;
}

export async function assertSafeUrl(input: string): Promise<URL> {
  try {
    if (
      typeof input !== "string" ||
      input.length === 0 ||
      input.length > 2048
    ) {
      throw new UnsafeUrlError();
    }

    const url = new URL(input);
    const authority = input.match(/^https:\/\/([^/?#]*)/i)?.[1];
 
    if (!authority || authority.includes("@")) {
  throw new UnsafeUrlError();
}

    if (
      url.protocol !== "https:" ||
      !url.hostname ||
      url.username ||
      url.password
    ) {
      throw new UnsafeUrlError();
    }

    const hostname = url.hostname
      .toLowerCase()
      .replace(/\.$/, "")
      .replace(/^\[|\]$/g, "");

    if (
      hostname === "localhost" ||
      hostname.endsWith(".localhost") ||
      hostname.endsWith(".local") ||
      hostname.endsWith(".internal") ||
      hostname.endsWith(".test") ||
      hostname.endsWith(".invalid")
    ) {
      throw new UnsafeUrlError();
    }
    const family = isIP(hostname);

    if (family !== 0) {
      if (addressIsUnsafe(hostname)) {
        throw new UnsafeUrlError();
      }

      return url;
    }
    if (!hostname.includes(".")) {
      throw new UnsafeUrlError();
    }

    // Reject malformed numeric IP representations.
    if (/^[0-9.]+$/.test(hostname)) {
      throw new UnsafeUrlError();
    }

    const addresses = await lookup(hostname, {
      all: true,
      order: "verbatim",
    });

    if (
      addresses.length === 0 ||
      addresses.some(({ address }) => addressIsUnsafe(address))
    ) {
      throw new UnsafeUrlError();
    }

    return url;
  } catch {
    throw new UnsafeUrlError();
  }
}

import {

const goodUrls = [
  "https://example.com",
  "https://www.google.com",
  "https://1.1.1.1",
  "https://[2606:4700:4700::1111]",
];

  "http://example.com",
  "https://localhost",
  "https://127.0.0.1",
  "https://10.0.0.1",
  "https://192.168.1.1",
  "https://172.16.0.1",
  "https://169.254.169.254",
  "https://[::1]",
  "https://[fc00::1]",
  "https://[fe80::1]",
  "https://user:password@example.com",
  "https://2130706433",
  "https://0177.0.0.1",
  "https://example.local",
  "https://@example.com",
  "https://intranet",
  "https://[3fff::1]",
  "https://[2001:db8::1]",
  "https://[::ffff:127.0.0.1]",
];

async function main() {
  console.log("\n--- SAFE URL TESTS ---");

  for (const url of goodUrls) {
    try {
      await assertSafeUrl(url);
      console.log("PASS:", url);
    } catch {
      console.log("FAIL: Safe URL rejected:", url);
      process.exitCode = 1;
    }
  }

  console.log("\n--- UNSAFE URL TESTS ---");

  for (const url of badUrls) {
    try {
      await assertSafeUrl(url);
      console.log("FAIL: Unsafe URL accepted:", url);
      process.exitCode = 1;
    } catch (error) {
      if (error instanceof UnsafeUrlError &&
          error.code === "UNSAFE_URL") {
        console.log("PASS: Rejected:", url);
      } else {
        console.log("FAIL: Unexpected error:", url);
        process.exitCode = 1;
      }
    }
  }

  if (process.exitCode === 1) {
    console.log("\nSome tests failed.");
  } else {
    console.log("\nAll URL safety tests passed.");
  }
}

main().catch(() => {
  console.error("URL safety test could not complete.");
  process.exitCode = 1;
});

import "dotenv/config";
import { verifySubmission } from "../src/services/verifier";

async function main() {
  const criteria = [
    {
      id: "criterion-1",
      text: "The homepage loads over HTTPS.",
    },
    {
      id: "criterion-2",
      text: "The homepage displays a visible main heading.",
    },
  ];

  const description =
    "I created the homepage and added a main heading. " +
    "The website is available at https://example.com.";

  console.log("Testing AI submission verification...");

  const result = await verifySubmission({
    criteria,
    description,
  });

  console.log("\nVerification result:");
  console.log(JSON.stringify(result, null, 2));

  const expectedIds = criteria.map((item) => item.id);
  const actualIds = result.criteriaResults.map(
    (item) => item.criterionId
  );

  if (
    expectedIds.length !== actualIds.length ||
    expectedIds.some((id) => !actualIds.includes(id)) ||
    new Set(actualIds).size !== actualIds.length
  ) {
    throw new Error("Criterion ID validation failed");
  }

  if (
    result.criteriaResults.some(
      (item) => item.status === "completed"
    )
  ) {
    throw new Error("Text-only verification returned completed");
  }

  console.log("\nPASS: Criterion IDs match.");
  console.log("PASS: No criterion was marked completed.");
}

main().catch((error) => {
  console.error("Verifier test failed:", error);
  process.exit(1);
});
